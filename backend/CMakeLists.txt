cmake_minimum_required(VERSION 3.20)
project(CareerPulse LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(SQLite3 REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(httplib CONFIG REQUIRED)
add_executable(careerpulse main.cpp)
target_link_libraries(careerpulse PRIVATE SQLite::SQLite3 nlohmann_json::nlohmann_json httplib::httplib bcrypt winhttp ws2_32 crypt32)
if(MSVC)
target_compile_options(careerpulse PRIVATE /utf-8)
endif()
